## About

The `<Sortable />` component is built on top of [@dnd-kit](https://dndkit.com)'s [sortable](https://docs.dndkit.com/presets/sortable) preset.

## Usage

```tsx
import {
  Sortable,
  SortableGrid,
  SortableItem,
  SortableItemTrigger,
  SortableList,
  SortableOverlay,
} from "@/components/ui/sortable";
```

```tsx
<Sortable>
  <SortableList>
    <SortableItem>
      <SortableItemTrigger />
    </SortableItem>
  </SortableList>
  <SortableGrid>
    <SortableItem>
      <SortableItemTrigger />
    </SortableItem>
  </SortableGrid>
  <SortableOverlay />
</Sortable>
```
