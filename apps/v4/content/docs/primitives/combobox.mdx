## About

The `<ComboboxPrimitive />` component is built on top of [cmdk](https://cmdk.paco.me), wrapping the library to deliver a more consistent API that aligns with Radix UI conventions.

## Anatomy

```tsx
import * as ComboboxPrimitive from "@/components/ui/combobox-primitive";

export default () => (
  <ComboboxPrimitive.Root>
    <ComboboxPrimitive.TagGroup>
      <ComboboxPrimitive.TagGroupItem>
        <ComboboxPrimitive.TagGroupItemRemove />
      </ComboboxPrimitive.TagGroupItem>
    </ComboboxPrimitive.TagGroup>

    <ComboboxPrimitive.Input />
    <ComboboxPrimitive.Clear />
    <ComboboxPrimitive.Anchor />
    <ComboboxPrimitive.Trigger />

    <ComboboxPrimitive.Portal>
      <ComboboxPrimitive.Content>
        <ComboboxPrimitive.List>
          <ComboboxPrimitive.Empty />
          <ComboboxPrimitive.Loading />

          <ComboboxPrimitive.Item>
            <ComboboxPrimitive.ItemIndicator />
          </ComboboxPrimitive.Item>

          <ComboboxPrimitive.Group>
            <ComboboxPrimitive.Item>
              <ComboboxPrimitive.ItemIndicator />
            </ComboboxPrimitive.Item>
          </ComboboxPrimitive.Group>

          <ComboboxPrimitive.Separator />
        </ComboboxPrimitive.List>
      </ComboboxPrimitive.Content>
    </ComboboxPrimitive.Portal>
  </ComboboxPrimitive.Root>
);
```
